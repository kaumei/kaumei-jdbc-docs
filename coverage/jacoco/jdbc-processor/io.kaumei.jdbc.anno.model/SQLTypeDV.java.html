package io.kaumei.jdbc.anno.model;

import java.sql.JDBCType;

public record SQLTypeDV(int sqlType) {

    public static SQLTypeDV of(JDBCType jdbcType) {
        return new SQLTypeDV(jdbcType.getVendorTypeNumber());
    }

    public static SQLTypeDV of(int sqlType) {
        return new SQLTypeDV(sqlType);
    }

}
