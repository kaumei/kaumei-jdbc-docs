package io.kaumei.jdbc.anno.store;

import io.kaumei.jdbc.anno.model.Visibility;

import javax.lang.model.type.TypeMirror;

public class ConverterBasis implements Converter {

    private final TypeMirror type;
    private final SourceDV source;
    private final Visibility visibility;

    protected ConverterBasis(TypeMirror type, SourceDV source, Visibility visibility) {
        this.type = type;
        this.source = source;
        this.visibility = visibility;
    }

    // ------------------------------------------------------------------------

    public TypeMirror type() {
        return this.type;
    }

    public SourceDV source() {
        return this.source;
    }

    public Visibility visibility() {
        return this.visibility;
    }

    // ------------------------------------------------------------------------

    @Override
    public String toString() {
        return getClass().getSimpleName() + "{" + this.type + "," + source + "," + visibility + "}";
    }

}
