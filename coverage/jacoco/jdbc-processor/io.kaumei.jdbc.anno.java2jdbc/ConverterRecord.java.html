package io.kaumei.jdbc.anno.java2jdbc;

import com.palantir.javapoet.CodeBlock;
import io.kaumei.jdbc.anno.gen.KaumeiLib;
import io.kaumei.jdbc.anno.gen.TargetMethod;
import io.kaumei.jdbc.anno.model.OptionalFlag;
import io.kaumei.jdbc.anno.model.Visibility;
import io.kaumei.jdbc.anno.store.SearchKey;
import io.kaumei.jdbc.anno.store.SourceDV;

import javax.lang.model.element.ExecutableElement;
import javax.lang.model.element.Name;
import javax.lang.model.type.TypeMirror;

final class ConverterRecord extends Java2JdbcConverter {

    record Placeholder(TypeMirror type, SourceDV source, Visibility visibility,
                       ExecutableElement method,
                       SearchKey other) implements Java2JdbcConverter.Placeholder {
        @Override
        public SearchKey otherPlaceholder() {
            return other;
        }

        @Override
        public Java2JdbcConverter resolve(Java2JdbcConverter converter) {
            return new ConverterRecord(this, converter);
        }
    }

    // ------------------------------------------------------------------------

    private final Name compName;
    private final Java2JdbcConverter converter;

    private ConverterRecord(Placeholder placeholder, Java2JdbcConverter converter) {
        super(placeholder.type, placeholder.source, placeholder.visibility.and(converter.visibility()));
        this.compName = placeholder.method.getSimpleName();
        this.converter = converter;
    }

    @Override
    public void setParameter(TargetMethod builder, CodeBlock paramAccess, CodeBlock columnIndex, OptionalFlag optional) {
        builder.addComment("ConverterRecord", "type", this.type(), "optional", optional);
        var localName = builder.tempVarName(paramAccess);
        if (optional.isNonNull()) {
            builder.beginControlFlow("if ($L == null)", paramAccess);
            builder.addStatement(KaumeiLib.throwNullPointerException(paramAccess));
            builder.endControlFlow();
            builder.addStatement("var $L = $L.$N()", localName, paramAccess, compName);
            converter.setParameter(builder, localName, columnIndex, optional);
        } else if (this.converter.type().getKind().isPrimitive()) {
            builder.beginControlFlow("if ($L == null)", paramAccess);
            builder.addStatement(builder.setNull(this.converter.type(), columnIndex));
            builder.nextControlFlow("else");
            builder.addStatement("var $L = $L.$N()", localName, paramAccess, compName);
            converter.setParameter(builder, localName, columnIndex, OptionalFlag.NON_NULL);
            builder.endControlFlow();
        } else {
            builder.addStatement("var $L = $L == null ? null : $L.$N()", localName, paramAccess, paramAccess, compName);
            converter.setParameter(builder, localName, columnIndex, optional);
        }
    }
}
