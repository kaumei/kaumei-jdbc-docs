package io.kaumei.jdbc.anno.msg;

import org.jspecify.annotations.Nullable;

import java.util.Iterator;
import java.util.NoSuchElementException;
import java.util.function.Supplier;

import static java.util.Objects.requireNonNull;

final class MessagesEmptyImpl implements Msg.Messages {

    static final Msg.Messages EMPTY_LIST = new MessagesEmptyImpl();

    @Override
    public String toString() {
        return "[]";
    }

    // interface Iterable<Msg.Message> ----------------------------------------

    @Override
    public Iterator<Msg.Message> iterator() {
        return EMPTY_ITERATOR;
    }

    // interface Msg.Messages -------------------------------------------------

    @Override
    public boolean hasMessages() {
        return false;
    }

    @Override
    public boolean hasNoMessages() {
        return true;
    }

    @Override
    public String join(String sep) {
        return "";
    }

    @Override
    public void join(StringBuilder sb, String sep) {
    }

    @Override
    public void requireEmpty() {
        // it is empty, that's what is expected, nothing to do
    }

    @Override
    public Msg.Messages requireNoneEmpty() {
        throw new IllegalStateException("List is empty");
    }

    @Override
    public <T> Msg.Result<T> build(T value) {
        return new MessageResultImpl<>(requireNonNull(value, "value"));
    }

    @Override
    public <T> Msg.Result<T> build(Supplier<T> func) {
        return new MessageResultImpl<>(requireNonNull(requireNonNull(func, "func").get(), "func.get"));
    }

    @Override
    public <T> Msg.NullableResult<T> buildOpt(@Nullable T value) {
        return new MessageResultNullableImpl<>(value);
    }

    // ------------------------------------------------------------------------

    private static final Iterator<Msg.Message> EMPTY_ITERATOR = new Iterator<>() {

        @Override
        public boolean hasNext() {
            return false;
        }

        @Override
        public Msg.Message next() {
            throw new NoSuchElementException();
        }
    };

}
