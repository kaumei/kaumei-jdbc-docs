package io.kaumei.jdbc.tx;

public interface KaumeiTxSynchronization {

    enum BeforeStatus {
        COMMITTING,
        ROLLING_BACK
    }

    enum AfterStatus {
        COMMITTED,
        ROLLED_BACK,
        UNKNOWN
    }

    // ------------------------------------------------------------------------

    default BeforeStatus beforeCompletion(BeforeStatus status) {
        return status;
    }

    default void afterCompletion(AfterStatus status) {
    }

}
